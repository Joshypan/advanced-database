"""Pets data layer backed by MongoDB through PyMongo, with indexes.

The application uses string IDs; stored _id and owner_id values are ObjectIds.
PyMongo: https://www.mongodb.com/docs/languages/python/pymongo-driver/current/
ObjectId: https://pymongo.readthedocs.io/en/stable/api/bson/objectid.html
Indexes: https://www.mongodb.com/docs/manual/indexes/

Chapter 14 change: setup_database() now calls ensure_indexes(), which creates
the indexes listed in INDEXES. MongoDB always indexes _id on its own.
"""
import os
from pymongo import MongoClient
from pymongo.server_api import ServerApi
from bson.errors import InvalidId

from bson.objectid import ObjectId

try:
    from mongita import MongitaClientMemory
except ImportError:  # pragma: no cover - some mongita installs may omit this helper
    MongitaClientMemory = None

try:
    import pytest
except ImportError:  # pragma: no cover - only needed when running the tests
    pytest = None

client = None
db = None
owners_collection = None
pets_collection = None

# Each entry: (collection name, key field, index name, extra options).
# create_index is idempotent: asking for an index that already exists with the
# same definition does nothing. An explicit name keeps the names predictable
# in mongosh and in explain output.
INDEXES = [
    ("pets", "owner_id", "pets_owner_id", {}),
    ("pets", "name", "pets_name", {}),
    ("pets", "type", "pets_type", {}),
    ("owners", "name", "owners_name", {}),
]


class NotFoundError(LookupError):
    pass


class ConstraintError(ValueError):
    pass


def initialize(database_name="pets", client_factory=None):
    global client, db, owners_collection, pets_collection

    close_connection()
    if client_factory is None:
        uri = os.environ.get("MONGODB_URI")
        if not uri:
            raise RuntimeError("Set MONGODB_URI before starting the application.")
        client = MongoClient(
            uri, server_api=ServerApi("1"), serverSelectionTimeoutMS=10000
        )
        try:
            # Force a connection now, so a bad configuration fails at startup.
            client.admin.command("ping")
        except Exception:
            close_connection()
            raise
    else:
        # Tests can supply a local memory store without contacting Atlas.
        client = client_factory()
    db = client[database_name]
    owners_collection = db.owners
    pets_collection = db.pets


def test_initialize_sets_globals():
    initialize("pytest_initialize", client_factory=MongitaClientMemory)
    assert client is not None
    assert db is not None
    assert owners_collection is not None
    assert pets_collection is not None
    close_connection()


def ensure_indexes():
    """Create the indexes in INDEXES and return the names that were requested.

    Run this against a real MongoDB server such as Atlas. Errors, such as a
    missing permission, are not caught here: they should be visible.
    """
    collections = {"owners": owners_collection, "pets": pets_collection}
    created = []
    for collection_name, field, index_name, options in INDEXES:
        collections[collection_name].create_index(
            [(field, 1)], name=index_name, **options
        )
        created.append(index_name)
    return created


def setup_database(database_name="pets", client_factory=None):
    """Connect, check read access, and make sure the indexes exist.

    Collection handles do not create stored collections. MongoDB creates them
    on the first insert unless they are explicitly created beforehand.
    Creating an index on a missing collection creates the collection, which is
    fine here: it stays empty until the first insert. No records are cleared
    or seeded.
    """
    initialize(database_name, client_factory=client_factory)
    owners_collection.count_documents({})
    pets_collection.count_documents({})
    ensure_indexes()


def test_setup_database_opens_collections_and_creates_indexes(monkeypatch):
    # Mongita cannot create named indexes, so stand in for ensure_indexes().
    calls = []
    monkeypatch.setitem(globals(), "ensure_indexes", lambda: calls.append("called"))
    setup_database("pytest_setup", client_factory=MongitaClientMemory)
    assert calls == ["called"]
    assert owners_collection.count_documents({}) == 0
    assert pets_collection.count_documents({}) == 0
    close_connection()


def test_ensure_indexes_requests_each_index(monkeypatch):
    from unittest.mock import MagicMock

    owners, pets = MagicMock(), MagicMock()
    monkeypatch.setitem(globals(), "owners_collection", owners)
    monkeypatch.setitem(globals(), "pets_collection", pets)
    assert ensure_indexes() == [name for _, _, name, _ in INDEXES]
    pets.create_index.assert_any_call([("owner_id", 1)], name="pets_owner_id")
    pets.create_index.assert_any_call([("name", 1)], name="pets_name")
    pets.create_index.assert_any_call([("type", 1)], name="pets_type")
    owners.create_index.assert_called_once_with([("name", 1)], name="owners_name")


def close_connection():
    global client, db, owners_collection, pets_collection

    if client is not None:
        try:
            client.close()
        except Exception:
            pass
    client = None
    db = None
    owners_collection = None
    pets_collection = None


def test_close_connection_resets_globals():
    initialize("pytest_close", client_factory=MongitaClientMemory)
    close_connection()
    assert client is None
    assert db is None
    assert owners_collection is None
    assert pets_collection is None


def _normalize_age(value):
    if value is None or (isinstance(value, str) and not value.strip()):
        return 0
    try:
        age = int(value)
    except (TypeError, ValueError) as exc:
        raise ValueError("Age must be a whole number.") from exc
    if age < 0:
        raise ValueError("Age must be non-negative.")
    return age


def _require_text(value, field_name):
    text = (value or "").strip()
    if text == "":
        raise ValueError(f"{field_name} is required.")
    return text


def _to_object_id(value, field_name="id"):
    # A URL/form supplies text; an _id filter needs the stored ObjectId type.
    try:
        return ObjectId(str(value))
    except (InvalidId, TypeError) as exc:
        raise ValueError(f"{field_name} must be a valid ObjectId string.") from exc


def _require_owner(owner_id):
    object_id = _to_object_id(owner_id, "owner_id")
    owner = owners_collection.find_one({"_id": object_id})
    if owner is None:
        raise ConstraintError("owner_id does not reference an existing owner.")
    return object_id


def _require_existing_pet(id):
    object_id = _to_object_id(id, "pet id")
    pet = pets_collection.find_one({"_id": object_id})
    if pet is None:
        raise NotFoundError("pet not found.")
    return object_id, pet


def _require_existing_owner(id):
    object_id = _to_object_id(id, "owner id")
    owner = owners_collection.find_one({"_id": object_id})
    if owner is None:
        raise NotFoundError("owner not found.")
    return object_id, owner


def _normalize_pet_data(data):
    owner_id = data.get("owner_id")
    if (owner_id or "").strip() == "":
        raise ValueError("owner_id is required.")

    return {
        "name": _require_text(data.get("name"), "name"),
        "type": _require_text(data.get("type"), "type"),
        "age": _normalize_age(data.get("age")),
        "owner_id": _require_owner(owner_id),
    }


def _normalize_owner_data(data):
    return {
        "name": _require_text(data.get("name"), "name"),
        "city": (data.get("city") or "").strip() or None,
        "type_of_home": (data.get("type_of_home") or "").strip() or None,
    }


def pet_to_dict(pet):
    # Keep BSON identifiers inside this layer; the app receives ordinary strings.
    return {
        "id": str(pet["_id"]),
        "name": pet["name"],
        "type": pet["type"],
        "age": pet["age"],
        "owner_id": str(pet["owner_id"]),
    }


def test_pet_to_dict():
    sample = {
        "_id": ObjectId("67d8c61b5180a31695e90744"),
        "name": "meercat",
        "type": "mammal",
        "age": 2,
        "owner_id": ObjectId("67d8c61b5180a31695e90745"),
    }
    converted = pet_to_dict(sample)
    assert converted["id"] == "67d8c61b5180a31695e90744"
    assert converted["owner_id"] == "67d8c61b5180a31695e90745"
    assert converted["name"] == "meercat"
    assert converted["type"] == "mammal"
    assert converted["age"] == 2

        "type_of_home": owner.get("type_of_home"),

def _seed_test_database(database_name="pytest_seed"):
    initialize(database_name, client_factory=MongitaClientMemory)

    owners_collection.delete_many({})
    pets_collection.delete_many({})

    owners = [
        {"name": "greg", "city": "Portland", "type_of_home": "condo"},
        {"name": "david", "city": "Seattle", "type_of_home": "farm"},
    ]
    owner_ids = {}
    for owner in owners:
        owner_id = create_owner(owner)
        owner_ids[owner["name"]] = owner_id

    pets = [
        {"name": "dorothy", "type": "dog", "age": 9, "owner_id": owner_ids["greg"]},
        {"name": "suzy", "type": "mouse", "age": 9, "owner_id": owner_ids["greg"]},
        {"name": "casey", "type": "dog", "age": 9, "owner_id": owner_ids["greg"]},
        {"name": "heidi", "type": "cat", "age": 15, "owner_id": owner_ids["david"]},
    ]
    for pet in pets:
        create_pet(pet)

    return owner_ids


def get_pets():
    return [pet_to_dict(pet) for pet in pets_collection.find()]


def test_get_pets():
    owner_ids = _seed_test_database()
    pets = get_pets()
    assert type(pets) is list
    assert len(pets) >= 1
    assert type(pets[0]) is dict
    for key in ["id", "name", "type", "age", "owner_id"]:
        assert key in pets[0]
    assert type(pets[0]["id"]) is str
    assert type(pets[0]["owner_id"]) is str
    assert pets[0]["owner_id"] in owner_ids.values()

def delete_owner(id):
    object_id, _ = _require_existing_owner(id)
    # This lookup by owner_id is the query the pets_owner_id index speeds up.
    # Without that index MongoDB examines every pet document to answer it.
    pet = pets_collection.find_one({"owner_id": object_id})
    if pet is not None:
        raise ConstraintError(
            "Cannot delete this owner because they have pets. Please delete their pets first."
        )

    owners_collection.delete_one({"_id": object_id})

def test_delete_owner_then_pet_succeeds():
    _seed_test_database()
    owner_id = create_owner({"name": "solo", "city": "Akron", "type_of_home": "house"})
    pet_id = create_pet(
        {"name": "onepet", "age": 3, "type": "cat", "owner_id": owner_id}
    )

    with pytest.raises(ConstraintError):
        delete_owner(owner_id)

explain-queries.py

"""Show the indexes and query plans for the Pets application's lookups.

Run this against Atlas after setting MONGODB_URI. It never inserts, updates,
or deletes anything. It prints, for each query, whether MongoDB used an index
(IXSCAN) or read the whole collection (COLLSCAN), and how many documents it
examined to return how many.

Explain output is a nested document whose exact layout can differ between
MongoDB server versions, so this script searches for the stage names instead
of relying on one fixed path.
"""
from bson.objectid import ObjectId

import database


def find_stages(plan, found=None):
    """Collect every 'stage' name found anywhere inside an explain plan."""
    if found is None:
        found = []
    if isinstance(plan, dict):
        if "stage" in plan:
            found.append(plan["stage"])
        for value in plan.values():
            find_stages(value, found)
    elif isinstance(plan, list):
        for item in plan:
            find_stages(item, found)
    return found


def explain(collection, query):
    return collection.find(query).explain()


def describe(label, collection, query):
    result = explain(collection, query)
    stages = find_stages(result.get("queryPlanner", {}).get("winningPlan", {}))
    stats = result.get("executionStats", {})
    print(label)
    print("  query:", query)
    print("  plan stages:", " -> ".join(stages) or "(none found)")
    if "COLLSCAN" in stages:
        print("  result: reads the whole collection")
    elif "IXSCAN" in stages:
        print("  result: uses an index")
    if stats:
        print("  documents examined:", stats.get("totalDocsExamined"))
        print("  keys examined:", stats.get("totalKeysExamined"))
        print("  documents returned:", stats.get("nReturned"))
    print()


def main():
    try:
        database.setup_database("pets")
        for collection in (database.owners_collection, database.pets_collection):
            print(f"Indexes on {collection.name}:")
            for name, info in collection.index_information().items():
                print("  ", name, info.get("key"))
            print()

        sample_owner = database.owners_collection.find_one()
        owner_id = sample_owner["_id"] if sample_owner else ObjectId()
        sample_pet = database.pets_collection.find_one()
        pet_name = sample_pet["name"] if sample_pet else "Casey"
        pet_type = sample_pet["type"] if sample_pet else "dog"
        owner_name = sample_owner["name"] if sample_owner else "Alex"

        describe("Pet lookup by identifier (automatic _id index)",
                 database.pets_collection, {"_id": sample_pet["_id"] if sample_pet else ObjectId()})
        describe("Does this owner still have pets? (used when deleting an owner)",
                 database.pets_collection, {"owner_id": owner_id})
        describe("Pets by name", database.pets_collection, {"name": pet_name})
        describe("Pets by type", database.pets_collection, {"type": pet_type})
        describe("Owners by name", database.owners_collection, {"name": owner_name})
        describe("Pets by age (no index on age)",
                 database.pets_collection, {"age": {"$gte": 0}})
    finally:
        database.close_connection()


if __name__ == "__main__":
    main()

explore-imdb-indexes.py

"""Time IMDb queries in Atlas before and after building indexes.

Usage:
    python3 explore-imdb-indexes.py [--database imdb]

Run import-imdb-mongo.py first. For each experiment this script prints the
query, the plan stage MongoDB chose (COLLSCAN or IXSCAN), how many documents
MongoDB examined, how many it returned, and the best of three timings. Then it
builds the index for that experiment, prints how long the build took, and
measures the same query again.

It creates only indexes whose names start with idx_, and it drops those at the
start, so you can run it again. It never changes the data.

Your numbers will not match anyone else's. They depend on your cluster tier,
your network, and whether Atlas already has the data in memory. Record your
own results in the table in the chapter exercise.
"""
import argparse
import os
import time

from pymongo import ASCENDING, DESCENDING, MongoClient
from pymongo.server_api import ServerApi

# Each experiment: (heading, collection, filter, sort, index keys, index name).
# sort is None when the query has no ordering.
EXPERIMENTS = [
    (
        "One title, one field",
        "titles",
        {"primary_title": "The Godfather"},
        None,
        [("primary_title", ASCENDING)],
        "idx_titles_primary_title",
    ),
    (
        "Two conditions",
        "titles",
        {"title_type": "movie", "start_year": 1994},
        None,
        [("title_type", ASCENDING), ("start_year", ASCENDING)],
        "idx_titles_type_year",
    ),
    (
        "Top ten by votes",
        "ratings",
        {},
        [("num_votes", DESCENDING)],
        [("num_votes", DESCENDING)],
        "idx_ratings_num_votes",
    ),
]


def connect(database_name):
    uri = os.environ.get("MONGODB_URI")
    if not uri:
        raise SystemExit("Set MONGODB_URI before running this script.")
    client = MongoClient(uri, server_api=ServerApi("1"), serverSelectionTimeoutMS=10000)
    client.admin.command("ping")
    return client, client[database_name]


def stages(plan):
    """Collect every stage name in a winning plan, from the top down."""
    found = []
    while plan:
        found.append(plan["stage"])
        plan = plan.get("inputStage")
    return found


def explain(collection, query, sort):
    cursor = collection.find(query)
    if sort:
        cursor = cursor.sort(sort)
    if sort:
        cursor = cursor.limit(10)
    report = cursor.explain()
    winning = report["queryPlanner"]["winningPlan"]
    executed = report["executionStats"]
    return (
        " > ".join(stages(winning)),
        executed["totalDocsExamined"],
        executed["nReturned"],
    )


def best_of_three(collection, query, sort):
    best = None
    for _ in range(3):
        cursor = collection.find(query)
        if sort:
            cursor = cursor.sort(sort).limit(10)
        started = time.perf_counter()
        list(cursor)
        elapsed = (time.perf_counter() - started) * 1000
        best = elapsed if best is None else min(best, elapsed)
    return best


def drop_experiment_indexes(database):
    for name in ("titles", "ratings"):
        collection = database[name]
        for index in collection.list_indexes():
            if index["name"].startswith("idx_"):
                collection.drop_index(index["name"])


def measure(label, collection, query, sort):
    plan, examined, returned = explain(collection, query, sort)
    milliseconds = best_of_three(collection, query, sort)
    print(f"  {label}")
    print(f"    plan: {plan}")
    print(f"    documents examined: {examined:,}   returned: {returned:,}")
    print(f"    best of three: {milliseconds:,.1f} ms")


def run(database_name):
    client, database = connect(database_name)
    try:
        drop_experiment_indexes(database)
        for heading, name, query, sort, keys, index_name in EXPERIMENTS:
            collection = database[name]
            print(f"\n{heading}: {name}.find({query})" + (f" sorted by {sort}" if sort else ""))
            measure("before the index", collection, query, sort)
            started = time.perf_counter()
            collection.create_index(keys, name=index_name)
            print(f"  built {index_name} in {time.perf_counter() - started:.1f} seconds")
            measure("after the index", collection, query, sort)
    finally:
        client.close()


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Time IMDb queries before and after indexes.")
    parser.add_argument("--database", default="imdb")
    args = parser.parse_args()
    run(args.database)

import-imdb-mongo.py

"""Load IMDb title.basics and title.ratings into MongoDB Atlas, in chunks.

Usage:
    python3 import-imdb-mongo.py [--data-dir data] [--database imdb] [--replace]

Reads the same two compressed files the Chapter 9 loader uses
(title.basics.tsv.gz and title.ratings.tsv.gz), one row at a time, so memory
use stays small. Documents go to Atlas in batches of 10,000 with insert_many.

Collections: titles and ratings. The IMDb identifier (tconst, such as
tt0000001) is stored as the document's _id, so MongoDB's automatic _id index
plays the role of the SQLite primary key. No other indexes are created here.
Building them is the exercise in Chapter 14.

Set MONGODB_URI first, the same way as for the pets application.
The IMDb files are for personal, non-commercial use. Do not redistribute them
or commit them to a public repository.
"""
import argparse
import csv
import gzip
from itertools import islice
import os
from pathlib import Path
import time

from pymongo import MongoClient
from pymongo.server_api import ServerApi

BATCH_SIZE = 10_000

# (file name, collection, [(field, kind), ...]). The first field becomes _id.
FILES = [
    (
        "title.basics.tsv.gz",
        "titles",
        [
            ("_id", "text"),
            ("title_type", "text"),
            ("primary_title", "text"),
            ("original_title", "text"),
            ("is_adult", "int"),
            ("start_year", "int"),
            ("end_year", "int"),
            ("runtime_minutes", "int"),
            ("genres", "text"),
        ],
    ),
    (
        "title.ratings.tsv.gz",
        "ratings",
        [("_id", "text"), ("average_rating", "real"), ("num_votes", "int")],
    ),
]


def convert(value, kind):
    """IMDb writes a missing value as \\N. Convert it to None, and numbers to numbers."""
    if value == "\\N":
        return None
    if kind == "int":
        return int(value)
    if kind == "real":
        return float(value)
    return value


def documents(path, fields):
    with gzip.open(path, "rt", encoding="utf-8", newline="") as handle:
        reader = csv.reader(handle, delimiter="\t", quoting=csv.QUOTE_NONE)
        next(reader)  # header line
        for record in reader:
            yield {
                name: convert(value, kind)
                for (name, kind), value in zip(fields, record)
            }


def load(collection, path, fields, limit=None):
    source = documents(path, fields)
    if limit is not None:
        source = islice(source, limit)
    total = 0
    while True:
        batch = list(islice(source, BATCH_SIZE))
        if not batch:
            break
        collection.insert_many(batch, ordered=False)
        before = total
        total += len(batch)
        if total // 500_000 > before // 500_000:
            print(f"  {collection.name}: {total:,} documents", flush=True)
    print(f"  {collection.name}: {total:,} documents in total")
    return total


def connect():
    uri = os.environ.get("MONGODB_URI")
    if not uri:
        raise SystemExit("Set MONGODB_URI before running this script.")
    client = MongoClient(uri, server_api=ServerApi("1"), serverSelectionTimeoutMS=10000)
    client.admin.command("ping")
    return client


def import_all(data_dir, database_name, replace, limit=None):
    if limit is not None and limit < 1:
        raise SystemExit("--limit must be positive.")
    for filename, _, _ in FILES:
        if not (Path(data_dir) / filename).is_file():
            raise SystemExit(f"Missing input: {Path(data_dir) / filename}")
    client = connect()
    try:
        database = client[database_name]
        existing = set(database.list_collection_names())
        for _, collection_name, _ in FILES:
            if collection_name in existing:
                if not replace:
                    raise SystemExit(
                        f"{database_name}.{collection_name} already exists. "
                        "Use --replace to rebuild it."
                    )
                database.drop_collection(collection_name)
        for filename, collection_name, fields in FILES:
            started = time.perf_counter()
            print(f"Loading {filename}")
            load(database[collection_name], Path(data_dir) / filename, fields, limit)
            print(f"  {time.perf_counter() - started:.1f} seconds")
    finally:
        client.close()


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Load IMDb files into MongoDB Atlas.")
    parser.add_argument("--data-dir", default="data")
    parser.add_argument("--database", default="imdb")
    parser.add_argument(
        "--replace",
        action="store_true",
        help="drop the titles and ratings collections first if they exist",
    )
    parser.add_argument("--limit", type=int, help="maximum documents per collection")
    args = parser.parse_args()
    import_all(args.data_dir, args.database, args.replace, args.limit)
