"""Back up the exercise schema and verify a fresh, separate restore."""
import hashlib
import os
from pathlib import Path
import shutil
import subprocess
import uuid
import psycopg
from psycopg import sql
from psycopg.conninfo import make_conninfo, conninfo_to_dict
uri=os.environ['DATABASE_URL']
restore_db='imdb_restore_'+uuid.uuid4().hex[:10]
folder=Path('.private-backups'); folder.mkdir(mode=0o700,exist_ok=True)
archive=folder/(restore_db+'.dump')
def binary(name):
    found=shutil.which(name)
    if not found: raise RuntimeError('Install PostgreSQL client tools: missing '+name)
    return found

def fingerprint(connection,table):
    digest=hashlib.sha256();count=0
    with connection.cursor(name='backup_check') as cur:
        cur.execute(sql.SQL('SELECT * FROM imdb_course.{} ORDER BY title_id').format(sql.Identifier(table)))
        for row in cur:
            digest.update((repr(row)+'\n').encode());count+=1
    return count,digest.hexdigest()
# Use a private environment variable, rather than putting a connection URI in process arguments.
env=dict(os.environ)
# libpq environment fields are individual values, not an expandable URI.
for key,value in conninfo_to_dict(uri).items():
    env['PG'+('DATABASE' if key=='dbname' else key.upper())]=value
subprocess.run([binary('pg_dump'),'--format=custom','--schema=imdb_course','--file',str(archive)],env=env,check=True)
with psycopg.connect(make_conninfo(uri,dbname='postgres'),autocommit=True) as admin:
    admin.execute(sql.SQL('CREATE DATABASE {}').format(sql.Identifier(restore_db)))
restore_uri=make_conninfo(uri,dbname=restore_db)
subprocess.run([binary('pg_restore'),'--dbname',restore_db,'--exit-on-error','--single-transaction','--no-owner','--no-privileges',str(archive)],env=dict(env,PGDATABASE=restore_db),check=True)
with psycopg.connect(uri) as source, psycopg.connect(restore_uri) as target:
    tables=['titles','ratings']
    if source.execute("SELECT to_regclass('imdb_course.movie_notes')").fetchone()[0]:tables.append('movie_notes')
    for table in tables:
        original=fingerprint(source,table); restored=fingerprint(target,table)
        assert original==restored,(table,'Restored data differ')
        print('PASS:',table,original[0],'rows, matching content digest')
    index_query="SELECT indexname,indexdef FROM pg_indexes WHERE schemaname='imdb_course' ORDER BY indexname"
    assert source.execute(index_query).fetchall()==target.execute(index_query).fetchall()
    print('PASS: restored index definitions match')
print('Archive:',archive)
print('Restore database:',restore_db)
print('Restore uses the administrative owner; application grants were deliberately omitted.')
print('Original database unchanged. Restore database kept for inspection.')

import os
import psycopg
from psycopg.conninfo import make_conninfo
uri=make_conninfo(os.environ['DATABASE_URL'],user='imdb_reader')
with psycopg.connect(uri) as conn:
    assert conn.execute('SELECT current_user').fetchone()[0]=='imdb_reader'
    assert conn.execute('SELECT count(*) FROM imdb_course.titles').fetchone()[0]>0
    print('PASS: dashboard login can read titles')
    conn.rollback()
    statements=[('UPDATE imdb_course.titles SET primary_title=primary_title WHERE false','update titles'),('CREATE ROLE imdb_permission_probe NOLOGIN','create a role')]
    for statement,label in statements:
        try:
            conn.execute(statement)
        except psycopg.errors.InsufficientPrivilege:
            print('PASS: denied '+label)
        else:
            raise AssertionError('Unexpected permission: '+label)
        finally:
            conn.rollback()

database.py

"""Small read-only data layer for the IMDb dashboard."""
import os
import psycopg
from psycopg.rows import dict_row

def query(statement, parameters=()):
    # Each call owns and closes its connection; Streamlit does not share it.
    with psycopg.connect(os.environ['DATABASE_URL'], row_factory=dict_row) as connection:
        return connection.execute(statement, parameters).fetchall()

def search_titles(text, minimum_votes=1000):
    return query("""
        SELECT t.title_id, t.primary_title, t.start_year,
               r.average_rating, r.num_votes
        FROM imdb_course.titles t LEFT JOIN imdb_course.ratings r USING (title_id)
        WHERE t.primary_title ILIKE %s AND coalesce(r.num_votes, 0) >= %s
        ORDER BY r.num_votes DESC, t.title_id LIMIT 100
    """, ('%'+text.strip()+'%', minimum_votes))

def yearly_summary(minimum_votes=1000):
    return query("""
        SELECT t.start_year, count(*) AS movies,
               round(avg(r.average_rating), 2) AS mean_rating
        FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
        WHERE t.start_year IS NOT NULL AND r.num_votes >= %s
        GROUP BY t.start_year ORDER BY t.start_year
    """, (minimum_votes,))

def ranked_movies(minimum_votes=1000):
    return query("""
        WITH ranked AS (
            SELECT t.title_id, t.primary_title, t.start_year, r.average_rating, r.num_votes,
                   dense_rank() OVER (
                       PARTITION BY t.start_year
                       ORDER BY r.average_rating DESC
                   ) AS year_rank
            FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
            WHERE t.start_year IS NOT NULL AND r.num_votes >= %s
        )
        SELECT * FROM ranked WHERE year_rank <= 3
        ORDER BY start_year DESC, year_rank, num_votes DESC, primary_title
        LIMIT 300
    """, (minimum_votes,))

def year_changes(minimum_votes=1000):
    return query("""
        WITH yearly AS (
            SELECT t.start_year, count(*) AS movies
            FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
            WHERE t.start_year IS NOT NULL AND r.num_votes >= %s
            GROUP BY t.start_year
        )
        SELECT start_year, movies,
               lag(start_year) OVER (ORDER BY start_year) AS previous_year,
               movies - lag(movies) OVER (ORDER BY start_year) AS change,
               sum(movies) OVER (
                   ORDER BY start_year ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
               ) AS running_total
        FROM yearly ORDER BY start_year
    """, (minimum_votes,))

def sample_summary():
    return query('SELECT count(*) AS titles, min(start_year) AS first_year, max(start_year) AS last_year FROM imdb_course.titles')[0]

def movies_by_year(year, minimum_votes=1000):
    return query("""SELECT t.title_id,t.primary_title,t.start_year,r.average_rating,r.num_votes
        FROM imdb_course.titles t LEFT JOIN imdb_course.ratings r USING(title_id)
        WHERE t.start_year=%s AND coalesce(r.num_votes,0)>=%s
        ORDER BY coalesce(r.num_votes,0) DESC,t.title_id LIMIT 100""",(int(year),minimum_votes))

def movies_by_rating(low, high, minimum_votes=1000):
    if not 0 <= low <= high <= 10:raise ValueError('Choose a rating range from 0 to 10.')
    return query("""SELECT t.title_id,t.primary_title,t.start_year,r.average_rating,r.num_votes
        FROM imdb_course.ratings r JOIN imdb_course.titles t USING(title_id)
        WHERE r.average_rating BETWEEN %s AND %s AND r.num_votes >= %s
        ORDER BY r.average_rating DESC,r.num_votes DESC,t.title_id LIMIT 100""",(low,high,minimum_votes))

def available_years():
    return [row['start_year'] for row in query('SELECT DISTINCT start_year FROM imdb_course.titles WHERE start_year IS NOT NULL ORDER BY start_year')]

import_imdb.py

"""Import prepared CSV files transactionally into the imdb_course schema."""
import argparse
import os
from pathlib import Path
import psycopg

HERE = Path(__file__).resolve().parent

def load(data_dir='data', replace=False):
    paths = [(table, Path(data_dir)/(table+'.csv')) for table in ['titles', 'ratings']]
    for table, path in paths:
        if not path.is_file():
            raise FileNotFoundError(path)
    with psycopg.connect(os.environ['DATABASE_URL']) as connection:
        exists = connection.execute("SELECT 1 FROM pg_namespace WHERE nspname='imdb_course'").fetchone()
        if exists and not replace:
            raise RuntimeError('imdb_course already exists. Use --replace only to replace this exercise schema.')
        if exists:
            connection.execute('DROP SCHEMA imdb_course CASCADE')
        connection.execute('CREATE SCHEMA imdb_course')
        connection.execute((HERE/'schema.sql').read_text())
        for table, path in paths:
            statement = psycopg.sql.SQL('COPY imdb_course.{} FROM STDIN WITH (FORMAT CSV, HEADER TRUE)').format(psycopg.sql.Identifier(table))
            with path.open('rb') as source, connection.cursor().copy(statement) as target:
                while block := source.read(1024*1024):
                    target.write(block)
        connection.execute((HERE/'indexes.sql').read_text())
        for table, _ in paths:
            count = connection.execute(psycopg.sql.SQL('SELECT count(*) FROM imdb_course.{}').format(psycopg.sql.Identifier(table))).fetchone()[0]
            print(f'{table}: {count:,} rows')
    print('Committed import, indexes and statistics.')

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--data-dir', default='data')
    parser.add_argument('--replace', action='store_true')
    args = parser.parse_args()
    load(args.data_dir, args.replace)

lecture-02-aggregates.sql

\set ON_ERROR_STOP on
-- One row per movie.
SELECT t.primary_title, t.start_year, r.average_rating, r.num_votes
FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
WHERE r.num_votes >= 1000 ORDER BY num_votes DESC, title_id LIMIT 20;
-- One row per year.
SELECT t.start_year, count(*) AS movies, round(avg(r.average_rating),2) AS mean_rating
FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
WHERE t.start_year IS NOT NULL AND r.num_votes >= 1000
GROUP BY t.start_year ORDER BY t.start_year;
-- Movie filter in WHERE; group filter in HAVING.
SELECT t.start_year, count(*) AS movies
FROM imdb_course.titles t JOIN imdb_course.ratings r USING (title_id)
WHERE t.start_year IS NOT NULL AND r.num_votes >= 1000 AND t.runtime_minutes >= 90
GROUP BY t.start_year HAVING count(*) >= 5 ORDER BY t.start_year;

lecture-03-windows.sql

\set ON_ERROR_STOP on
-- Deterministic tie fixture: dense ranks 1,1,2; row numbers 1,2,3.
WITH films(title,rating) AS (VALUES ('Alpha',8.0),('Beta',8.0),('Gamma',7.0))
SELECT title, rating, dense_rank() OVER (ORDER BY rating DESC) AS rating_rank,
 row_number() OVER (ORDER BY rating DESC,title) AS movie_number
FROM films ORDER BY rating DESC,title;
-- Missing-year fixture: changes NULL,-2; totals 3,4.
WITH yearly(year,movies) AS (VALUES (2000,3),(2002,1))
SELECT year,movies,lag(year) OVER (ORDER BY year) AS previous_year,
 movies-lag(movies) OVER (ORDER BY year) AS change,
 sum(movies) OVER (ORDER BY year ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total
FROM yearly ORDER BY year;
-- Same ranking operation on imported movies.
WITH ranked AS (
 SELECT t.primary_title,t.start_year,r.average_rating,r.num_votes,
 dense_rank() OVER (PARTITION BY t.start_year ORDER BY r.average_rating DESC) AS year_rank
 FROM imdb_course.titles t JOIN imdb_course.ratings r USING(title_id)
 WHERE t.start_year IS NOT NULL AND r.num_votes>=1000
)
SELECT * FROM ranked WHERE year_rank<=3
ORDER BY start_year DESC,year_rank,num_votes DESC,primary_title LIMIT 100;
-- A movie and the mean of its retained year group.
SELECT t.primary_title,t.start_year,r.average_rating,
 avg(r.average_rating) OVER (PARTITION BY t.start_year) AS year_mean,
 r.average_rating-avg(r.average_rating) OVER (PARTITION BY t.start_year) AS above_year_mean
FROM imdb_course.titles t JOIN imdb_course.ratings r USING(title_id)
WHERE t.start_year IS NOT NULL AND r.num_votes>=1000
ORDER BY t.start_year DESC,t.title_id LIMIT 100;

import streamlit as st

def choose_page():
    requested=st.query_params.get('page','Search')
    if requested=='Transactions':requested='Notes'
    if requested not in ['Search','Notes','Reports']:requested='Search'
    if st.session_state.get('_query_page')!=requested:
        st.session_state['page_navigation']=requested
        st.session_state['_query_page']=requested
    def navigate():
        chosen=st.session_state['page_navigation']
        st.query_params['page']=chosen
        st.session_state['_query_page']=chosen
    return st.sidebar.radio('Page',['Search','Notes','Reports'],key='page_navigation',on_change=navigate)

def linked_movies(rows):
    from urllib.parse import urlencode
    output=[]
    for row in rows:
        title_id=row.get('title_id',row.get('_id'))
        item=dict(row)
        item['movie_notes']='?'+urlencode({'page':'Notes','title_id':title_id,'title':row['primary_title']})
        output.append(item)
    return output

def show_movies(rows):
    if not rows:
        st.info('No matching movies with the current filters.')

prepare_csv.py

"""Convert local IMDb TSV archives to portable movie/ratings CSV files."""
import argparse
import csv
import gzip
from pathlib import Path


def prepare(data_dir, limit=100000, start_year=None, end_year=None):
    if limit < 0:
        raise ValueError('limit must be nonnegative; zero means all movies')
    if start_year is not None and end_year is not None and start_year > end_year:
        raise ValueError('start year must not exceed end year')
    data_dir = Path(data_dir)
    for name in ['title.basics.tsv.gz', 'title.ratings.tsv.gz']:
        if not (data_dir/name).is_file():
            raise FileNotFoundError(data_dir/name)
    ids = set()
    with gzip.open(data_dir/'title.basics.tsv.gz', 'rt', encoding='utf-8', newline='') as source, (data_dir/'titles.csv').open('w', encoding='utf-8', newline='') as target:
        reader = csv.DictReader(source, delimiter='\t', quoting=csv.QUOTE_NONE)
        writer = csv.writer(target)
        writer.writerow(['title_id', 'primary_title', 'start_year', 'runtime_minutes', 'genres'])
        for row in reader:
            if row['titleType'] != 'movie' or row['isAdult'] != '0':
                continue
            year = None if row['startYear'] == r'\N' else int(row['startYear'])
            if start_year is not None and (year is None or year < start_year):
                continue
            if end_year is not None and (year is None or year > end_year):
                continue
            ids.add(row['tconst'])
            writer.writerow([row['tconst'], row['primaryTitle'], '' if row['startYear'] == r'\N' else row['startYear'], '' if row['runtimeMinutes'] == r'\N' else row['runtimeMinutes'], '' if row['genres'] == r'\N' else row['genres']])
            if limit and len(ids) >= limit:
                break
    ratings = 0
    with gzip.open(data_dir/'title.ratings.tsv.gz', 'rt', encoding='utf-8', newline='') as source, (data_dir/'ratings.csv').open('w', encoding='utf-8', newline='') as target:
        reader = csv.DictReader(source, delimiter='\t', quoting=csv.QUOTE_NONE)
        writer = csv.writer(target)
        writer.writerow(['title_id', 'average_rating', 'num_votes'])
        for row in reader:
            if row['tconst'] in ids:
                writer.writerow([row['tconst'], row['averageRating'], row['numVotes']])
                ratings += 1
    print(f'Prepared {len(ids):,} movies and {ratings:,} matching ratings.')
    return len(ids), ratings

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--data-dir', default='data')
    parser.add_argument('--limit', type=int, default=0, help='maximum movies; 0 loads all eligible movies')
    parser.add_argument('--start-year', type=int, default=1975)
    parser.add_argument('--end-year', type=int, default=1999)
    args = parser.parse_args()
    prepare(args.data_dir, args.limit, args.start_year, args.end_year)

"""Create the dashboard login using an administrative DATABASE_URL."""
import os
import psycopg
from psycopg import sql
ROLE = 'imdb_reader'
with psycopg.connect(os.environ['DATABASE_URL']) as conn:
    if not conn.execute("SELECT 1 FROM pg_roles WHERE rolname = %s",(ROLE,)).fetchone():
        conn.execute(sql.SQL('CREATE ROLE {} LOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE NOREPLICATION NOBYPASSRLS').format(sql.Identifier(ROLE)))
    unsafe=conn.execute('SELECT rolsuper OR rolcreatedb OR rolcreaterole OR rolreplication OR rolbypassrls FROM pg_roles WHERE rolname=%s',(ROLE,)).fetchone()[0]
    memberships=conn.execute('SELECT 1 FROM pg_auth_members WHERE member=(SELECT oid FROM pg_roles WHERE rolname=%s)',(ROLE,)).fetchone()

    conn.execute('GRANT USAGE ON SCHEMA imdb_course TO imdb_reader')

#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")"
pg_bin="$(pg_config --bindir)"
pg_data="$PWD/.private-postgres"
mkdir -p "$pg_data"
chmod 700 "$pg_data"
exec 9>"$PWD/.private-postgres-start.lock"
flock 9
if [ ! -f "$pg_data/PG_VERSION" ]; then
    "$pg_bin/initdb" -D "$pg_data" -U student --auth-local=trust --auth-host=trust --encoding=UTF8 --locale=C.UTF-8
    cat >> "$pg_data/postgresql.conf" <<'CONF'
listen_addresses = '127.0.0.1'
port = 5433
unix_socket_directories = '/tmp'
CONF
fi
if ! "$pg_bin/pg_ctl" -D "$pg_data" status >/dev/null 2>&1; then
    # Keep the startup lock in this shell, not in the long-running server.
    "$pg_bin/pg_ctl" -D "$pg_data" -l "$pg_data/server.log" -w start 9>&-
fi
if ! psql 'postgresql://student@127.0.0.1:5433/postgres' -Atc "SELECT 1 FROM pg_database WHERE datname = 'imdb'" | grep -qx 1; then
    createdb -h 127.0.0.1 -p 5433 -U student imdb
fi
psql 'postgresql://student@127.0.0.1:5433/imdb' -c 'SELECT current_database(), current_user;'
